public class Program
    {
        static void Main(string[] args)
        {
            List<Inotificacao> listaNotificacoes = new List<Inotificacao>
            {
                new NotificacaoEmail(),
                new NotificacaoSMS(),
                new NotificacaoPush()
            };

            Console.WriteLine("--- DEMONSTRAÇÃO: POLIMORFISMO DE SOBRECRITA ---");
            foreach (Inotificacao notificacao in listaNotificacoes)
            {
                notificacao.Enviar();
            }

            Console.WriteLine();
            Console.WriteLine("--- DEMONSTRAÇÃO: POLIMORFISMO DE SOBRECARGA ---");
            NotificacaoEmail email = new NotificacaoEmail();

            email.Enviar();
            email.Enviar("relatorio.pdf");
        }
    }


    public class Inotificacao
    {
        public void Enviar();
    }


    public class NotificacaoEmail : Inotificacao
    {
        public void Enviar()
        {
            Console.WriteLine("Enviando notificação por email...");
        }

        public void Enviar(string anexo)
        {
            Console.WriteLine($"Enviando email com anexo: {anexo}");
        }
    }


    public class NotificacaoSMS : Inotificacao
    {
        public void Enviar()
        {
            Console.WriteLine("Enviando notificação por SMS...");
        }
    }

    public class NotificacaoPush : Inotificacao
    {
        public void Enviar()
        {
            Console.WriteLine("Enviando notificação push para o app...");
        }
    }
